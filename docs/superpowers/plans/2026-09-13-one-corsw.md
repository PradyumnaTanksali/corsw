```tsx
"use client";

import { useSyncExternalStore } from "react";
import { cn } from "@/lib/utils";

type Theme = "paper" | "schematic";
const THEMES: Theme[] = ["paper", "schematic"];

function subscribe(onChange: () => void) {
  const observer = new MutationObserver(onChange);
  observer.observe(document.documentElement, { attributes: true, attributeFilter: ["data-theme"] });
  return () => observer.disconnect();
}

const read = (): Theme =>
  document.documentElement.dataset.theme === "schematic" ? "schematic" : "paper";

export function ThemeToggle({ className }: { className?: string }) {
  const theme = useSyncExternalStore(subscribe, read, () => "paper");

  function choose(next: Theme) {
    document.documentElement.dataset.theme = next;
    try {
      localStorage.setItem("corsw-theme", next);
    } catch {
      // Private mode: the choice lasts for this page view only.
    }
  }

  return (
    <div role="group" aria-label="Theme" className={cn("flex items-baseline gap-1 print:hidden", className)}>
      {THEMES.map((t, i) => (
        <span key={t} className="flex items-baseline gap-1">
          {i > 0 && <span aria-hidden="true" className="text-ink-faint">/</span>}
          <button
            type="button"
            aria-pressed={theme === t}
            onClick={() => choose(t)}
            className="px-1 py-2 uppercase tracking-[0.22em] text-ink-faint transition-colors duration-150 hover:text-ink-muted focus-visible:outline focus-visible:outline-1 focus-visible:outline-accent aria-pressed:text-ink aria-pressed:underline aria-pressed:decoration-accent aria-pressed:underline-offset-4"
          >
            {t}
          </button>
        </span>
      ))}
    </div>
  );
}
```

- [ ] **Step 3: `lib/host.ts`**

```ts
const CANONICAL = "corsw.in";

export type HostDecision =
  | { kind: "redirect"; url: string }
  | { kind: "rewrite"; path: string; noindex: boolean }
  | { kind: "next"; noindex: boolean };

/**
 * corsw.in owns the *.corsw.in wildcard. Project subdomains are assigned to
 * their own Vercel projects and never reach this code; anything else does.
 */
export function decide(host: string, pathname: string, search: string): HostDecision {
  const name = host.toLowerCase();
  // Absolute origin + a path that always starts with "/" keeps the redirect on corsw.in.
  if (name === `www.${CANONICAL}`) return { kind: "redirect", url: `https://${CANONICAL}${pathname}${search}` };
  const noindex = name !== CANONICAL;
  if (name.endsWith(`.${CANONICAL}`) && pathname === "/") return { kind: "rewrite", path: "/demo", noindex };
  return { kind: "next", noindex };
}
```

```ts
import { NextResponse, type NextRequest } from "next/server";
import { decide } from "@/lib/host";

export function proxy(req: NextRequest) {
  const { pathname, search } = req.nextUrl;
  const d = decide(req.headers.get("host") ?? "", pathname, search);
  if (d.kind === "redirect") return NextResponse.redirect(new URL(d.url), 308);
  const res = d.kind === "rewrite" ? NextResponse.rewrite(new URL(d.path, req.url)) : NextResponse.next();
  if (d.noindex) res.headers.set("X-Robots-Tag", "noindex");
  return res;
}

export const config = {
  matcher: ["/((?!_next/static|_next/image|brand/|icon|apple-icon).*)"],
};
```
